# 极速入门

## 变量与数据类型

```py
# Python 变量不需要声明类型
name = "小明"           # 字符串
age = 25               # 整数
height = 1.75          # 浮点数
is_student = True      # 布尔值 (True/False，首字母大写！)
# 多变量赋值
x, y, z = 1, 2, 3
```

## 字符串

```py
name = "小明"
age = 25
# 字符串格式化 (Python 3.6+ 推荐)
msg = f"我叫{name}，今年{age}岁"  # ✅ f-string
msg2 = "我叫{}，今年{}岁".format(name, age)  # ✅ format
# 常用操作
len(name)           # 长度: 2
name.upper()        # "小明" → "小明"
name.replace("小", "老")  # "老明"
```

## 列表 (类似 JS 数组)

```py
fruits = ["苹果", "香蕉", "橙子"]

# 访问
fruits[0]           # "苹果"
fruits[-1]          # "橙子" (倒数第一个)

# 操作
fruits.append("葡萄")      # 添加
fruits.insert(1, "梨")     # 插入
fruits.remove("香蕉")       # 删除
fruits.pop()               # 弹出最后一个

# 切片 (Python特有！)
fruits[1:3]       # ["香蕉", "橙子"]
fruits[:2]        # ["苹果", "香蕉"]
fruits[::2]       # ["苹果", "橙子"] (步长2)
```

## 字典 (类似 JS 对象)

```py
person = {
    "name": "小明",
    "age": 25,
    "city": "北京"
}

# 访问
person["name"]          # "小明"
person.get("name")      # 同上，更安全
person.get("gender", "未知")  # 默认值

# 操作
person["age"] = 26      # 修改
person["gender"] = "男" # 添加
del person["age"]       # 删除
```

## 条件判断

```py
age = 20

if age < 18:
    print("未成年")

    print("成年人")
else:
    print("老年人")

# Python 用 elif，不是 else if ！
```

## 循环

```py
fruits = ["苹果", "香蕉", "橙子"]

# for 循环 (最常用)
for fruit in fruits:
    print(fruit)

# 带索引
for i, fruit in enumerate(fruits):
    print(f"{i}: {fruit}")

# range (类似 JS 的 for 循环)
for i in range(5):      # 0,1,2,3,4
    print(i)

for i in range(1, 6):   # 1,2,3,4,5
    print(i)

# while 循环
count = 0
while count < 3:
    print(count)
    count += 1
```

## 函数

```py
# 定义函数
def greet(name, greeting="你好"):
    return f"{greeting}，{name}!"

# 调用
result = greet("小明")           # "你好，小明！"
result = greet("小明", "早上好")  # "早上好，小明！"

# 参数可以是任意类型
def sum_all(*args):  # *args 收集多余参数 → 元组
    return sum(args)

sum_all(1, 2, 3, 4)  # 10

def info(**kwargs):  # **kwargs 收集关键字参数 → 字典
    return kwargs

info(name="小明", age=25)  # {'name': '小明', 'age': 25}
```

## 常用内置函数

```py
# 类型转换
int("123")      # 123
str(123)        # "123"
float("3.14")   # 3.14
list("abc")     # ["a", "b", "c"]

# 实用函数
len([1,2,3])    # 3
sum([1,2,3])    # 6
max([1,2,3])    # 3
min([1,2,3])    # 1
sorted([3,1,2]) # [1,2,3]
```

## 类 (OOP)

```py
class Person:
    def __init__(self, name, age):  # 构造函数
        self.name = name
        self.age = age
    
    def say_hello(self):
        return f"你好，我是{self.name}"

# 创建实例
p = Person("小明", 25)
print(p.say_hello())  # "你好，我是小明"
```
