
# GORM

GORM 是一款为 Go 语言 (Golang) 设计的开源 对象关系映射 (ORM) 库。

它的主要功能是让开发者能够通过操作 Go 语言中的结构体（Struct）来间接操作数据库（如 MySQL、PostgreSQL 等），从而避免编写复杂的原生 SQL 语句。

```go
# 安装 GORM 核心库
go get -u gorm.io/gorm

# 根据你使用的数据库，安装对应的驱动
go get -u gorm.io/driver/mysql      # MySQL
go get -u gorm.io/driver/postgres   # PostgreSQL
go get -u gorm.io/driver/sqlite     # SQLite
go get -u gorm.io/driver/mssql      # SQL Server
```

```go
package main

import (
    "fmt"
    "log"
    "time"

    "gorm.io/driver/mysql"
    "gorm.io/gorm"
)

type User struct {
    ID        uint
    Name      string
    Email     string *string // 使用指针，处理 NULL 值
    Age       uint8
    Birthday  *time.Time
    CreatedAt time.Time
    UpdatedAt time.Time
}

func main() {
    // 连接数据库
    dsn := "root:password@tcp(127.0.0.1:3306)/gorm_test?charset=utf8mb4&parseTime=True&loc=Local"
    db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{})
    if err != nil {
        log.Fatalf("数据库连接失败: %v", err)
    }
    defer func() { _ = db.Close() }()

    // 自动迁移数据库表结构
    err = db.AutoMigrate(&User{})
    if err != nil {
        log.Fatalf("迁移失败: %v", err)
    }

    // 创建记录
    user := User{
        Name:  "张三",
        Email: stringPtr("zhangsan@example.com"),
        Age:   25,
    }
    result := db.Create(&user)
    if result.Error != nil {
        log.Fatalf("创建记录失败: %v", result.Error)
    }
    fmt.Printf("创建成功，用户ID: %d\n", user.ID)

    // 查询记录
    var findUser User
    db.First(&findUser, user.ID)
    fmt.Printf("查询结果: %+v\n", findUser)

    // 更新记录
    db.Model(&findUser).Update("Name", "张三丰")
    fmt.Println("更新成功")

    // 删除记录
    db.Delete(&findUser)
    fmt.Println("删除成功")
}

func stringPtr(s string) *string {
    return &s
}
```
