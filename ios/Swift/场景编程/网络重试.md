# 网络重试

```swift
class NetworkManager {
    private var token = "expired_token"
    private var isRefreshing = false
    
    // 关键：存储等待中请求的数组
    private var retryQueue: [(String) -> Void] = []
    private let lock = NSLock()

    func request(url: String, completion: @escaping (String) -> Void) {
        let capturedToken = self.token // 记录发起请求时的 Token 版本

        // 1. 发起原始请求
        performRequest(url, token: token) { [weak self] result in
            guard let self = self else { return }
            
            if result == "401" {
                self.lock.lock() // 加锁防止多线程数据竞争

            // 【核心改进】：如果当前的 token 已经和发起请求时的 token 不一样了
            // 说明在 请求期间，已经有人刷新过 Token 了
            if self.token != capturedToken {
                print("✨ 检测到 Token 已被其他请求刷新，直接用新的重试")
                let newToken = self.token
                self.lock.unlock()
                self.request(url: url, completion: completion) // 直接重试
                return
            }
                
                // 2. 将当前请求的重试逻辑加入队列
                self.retryQueue.append({ newToken in
                    self.request(url: url, completion: completion)
                })
                
                // 3. 判断是否已经在刷新中
                if !self.isRefreshing {
                    self.isRefreshing = true
                    self.lock.unlock()
                    
                    self.startRefreshToken { newToken in
                        self.lock.lock()
                        self.token = newToken
                        self.isRefreshing = false
                        
                        // 4. 刷新成功，取出队列中所有请求并执行
                        let queue = self.retryQueue
                        self.retryQueue.removeAll()
                        self.lock.unlock()
                        
                        print("✅ 刷新成功，批量处理 \(queue.count) 个挂起的请求")
                        queue.forEach { $0(newToken) }
                    }
                } else {
                    print("⏳ 正在刷新中，请求已入队等待...")
                    self.lock.unlock()
                }
            } else {
                completion(result)
            }
        }
    }

    // 模拟刷新接口
    private func startRefreshToken(completion: @escaping (String) -> Void) {
        DispatchQueue.global().asyncAfter(deadline: .now() + 1) {
            completion("new_token")
        }
    }
    
    private func performRequest(_ url: String, token: String, completion: @escaping (String) -> Void) {
        completion(token == "expired_token" ? "401" : "Success Data")
    }
}
```
