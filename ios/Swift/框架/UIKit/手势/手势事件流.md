
# 手势事件流

- 注意 loggingTapGestureRecognizer.cancelsTouchesInView

```swift
/// UITapGestureRecognizer 日志输出
class LoggingTapGestureRecognizer: UITapGestureRecognizer {
    //
    override var state: UIGestureRecognizer.State {
        didSet {
            print("tap state:", state.rawValue)
        }
    }
    
    // touchesBegan
    override func touchesBegan(_ touches: Set<UITouch>, with event: UIEvent) {
        print("UITapGestureRecognizer_touchesBegan")
        super.touchesBegan(touches, with: event)
    }
    
    // touchesMoved
    override func touchesMoved(_ touches: Set<UITouch>, with event: UIEvent) {
        print("UITapGestureRecognizer_touchesMoved")
        super.touchesMoved(touches, with: event)
    }
    
    // touchesCancelled
    override func touchesCancelled(_ touches: Set<UITouch>, with event: UIEvent) {
            print("UITapGestureRecognizer_touchesCancelled")
        super.touchesCancelled(touches, with: event)
    }
    
    // touchesMoved
    override func touchesEnded(_ touches: Set<UITouch>, with event: UIEvent) {
        print("UITapGestureRecognizer_touchesEnded")
        super.touchesEnded(touches, with: event)
    }
}

class LoggingView:UIView {
    override func touchesBegan(_ touches: Set<UITouch>, with event: UIEvent?) {
        print("LoggingView_touchesBegan")
        super.touchesBegan(touches, with: event)
    }
    
    override func touchesEnded(_ touches: Set<UITouch>, with event: UIEvent?) {
        print("LoggingView_touchesEnded")
        super.touchesBegan(touches, with: event)
    }
    
    override func touchesMoved(_ touches: Set<UITouch>, with event: UIEvent?) {
        print("LoggingView_touchesMoved")
        super.touchesMoved(touches, with: event)
    }
    
    override func touchesCancelled(_ touches: Set<UITouch>, with event: UIEvent?) {
        print("LoggingView_touchesCancelled")
        super.touchesCancelled(touches, with: event)
    }
}

class DemoViewController: UIViewController, UIGestureRecognizerDelegate { 

    override func viewDidLoad() {        
        super.viewDidLoad()
        
        let _vv = LoggingView()
        let loggingTapGestureRecognizer = LoggingTapGestureRecognizer(target: self, action: #selector(handleTap(_:)))
        loggingTapGestureRecognizer.delegate = self
        loggingTapGestureRecognizer.cancelsTouchesInView = true
        _vv.addGestureRecognizer(loggingTapGestureRecognizer)
        view.addSubview(_vv)
        _vv.frame = .init(x: 0, y: 100, width: 100, height: 100)
        _vv.backgroundColor = .random
        
        
    }

    func gestureRecognizerShouldBegin(_ gestureRecognizer: UIGestureRecognizer) -> Bool {
        print("gestureRecognizer_gestureRecognizerShouldBegin")
        return true
    }
    
    func  gestureRecognizer(_ gestureRecognizer: UIGestureRecognizer, shouldRecognizeSimultaneouslyWith otherGestureRecognizer: UIGestureRecognizer) -> Bool {
        print("gestureRecognizer_shouldRecognizeSimultaneouslyWith")
        return true
    }
    
    func gestureRecognizer(_ gestureRecognizer: UIGestureRecognizer, shouldReceive touch: UITouch) -> Bool{
        print("gestureRecognizer_shouldReceive")
        return true
    }

}
```
