
# 简单示例代码 Publisher  Subscription Subscriber协作

```swift
import Combine
import Foundation

// 1. 定义 Subscriber (收货人)
class SimpleSubscriber: Subscriber {
    typealias Input = String
    typealias Failure = Never

    // 握手：收到订阅令牌（水管/快递单）
    func receive(subscription: Subscription) {
        print("1. 订阅者：已收到订阅，现在请求 2 个数据")
        // 关键：必须由订阅者发起请求，数据才会流动
        subscription.request(.max(2)) 
    }

    // 接收数据：处理发来的值
    func receive(_ input: String) -> Subscribers.Demand {
        print("3. 订阅者：收到数据 —— \(input)")
        return .none // 不再追加额外请求
    }

    // 结束：收到完成信号
    func receive(completion: Subscribers.Completion<Never>) {
        print("4. 订阅者：处理完毕，订单结束")
    }
}

// 2. 定义 Subscription (订阅令牌/快递单)
class SimpleSubscription<S: Subscriber>: Subscription where S.Input == String {
    private var subscriber: S?

    init(subscriber: S) {
        self.subscriber = subscriber
    }

    func request(_ demand: Subscribers.Demand) {
        print("2. 订阅令牌：收到 \(demand) 个需求量，开始发货...")
        // 模拟发送数据
        _ = subscriber?.receive("苹果")
        _ = subscriber?.receive("香蕉")
        _ = subscriber?.receive(completion: .finished)
    }

    func cancel() {
        subscriber = nil
    }
}

// 3. 定义 Publisher (发布者/发货员)
struct SimplePublisher: Publisher {
    typealias Output = String
    typealias Failure = Never

    func receive<S>(subscriber: S) where S : Subscriber, Failure == S.Failure, Output == S.Input {
        // 创建连接器（Subscription），并将它交给订阅者
        let subscription = SimpleSubscription(subscriber: subscriber)
        subscriber.receive(subscription: subscription)
    }
}

// --- 实际运行 ---
let publisher = SimplePublisher()
let subscriber = SimpleSubscriber()

print("开始订阅流程...")
publisher.subscribe(subscriber)

```
