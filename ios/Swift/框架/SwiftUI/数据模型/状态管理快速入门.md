import SwiftUI

struct ContentView: View {
    @State var state_title = "Hello, world!!"
    var body: some View {
        MyView(state_title: $state_title)
    }
}

// @State
// @Binding
struct MyView:View {
    @Binding var state_title:String
    var body: some View {
        VStack {
            Image(systemName: "globe")
                .imageScale(.large)
                .foregroundStyle(.tint)
            Text(state_title)
            Button("点击修改") {
                state_title = "小明"
            }
            Divider()
            StateObjectMainView()
            Divider()
            EnvironmentObjectRootView()
            Divider()
            EnvironmentMainView()
            Divider()
            if #available(iOS 17.0, *) {
                ObservableMainView()
            } else {
                // Fallback on earlier versions
            }
        }
        .padding()
    }
}

// ------------------------------------
// 用于 Class (引用类型)。
// @StateObject
// @ObservedObject

class DataModelStateObject:ObservableObject{
    @Published var count = 0
}

struct StateObjectMainView:View {
    @StateObject var model = DataModelStateObject()
    var body: some View {
        Text(String(model.count))
        StateObjectDetailView(model: model)
    }
}

struct StateObjectDetailView:View {
    @ObservedObject var model:DataModelStateObject
    var body: some View {
        Text(String(model.count))
        Button("点击修改") {
            model.count+=1
        }
    }
}

// ------------------------------------
// 跨层级传递，避免繁琐的参数传递。
// @EnvironmentObject

class DataModelEnvironmentObject: ObservableObject {
    @Published var count = 0
}

struct EnvironmentObjectRootView:View {
    @StateObject var settings = DataModelEnvironmentObject()
    @StateObject var settings1 = DataModelEnvironmentObject()
    var body: some View {
        VStack {
            Text("RootView")
            Text("count: \(settings.count)")
            Text("count: \(settings1.count)")
            EnvironmentObjectMainView()
                .environmentObject(settings)
                .environmentObject(settings1)
        }
    }
}

struct EnvironmentObjectMainView:View {
    @EnvironmentObject var settings:DataModelEnvironmentObject
    var body: some View {
        VStack {
            Text("MainView")
            Text("count: \(settings.count)")
            EnvironmentObjectSubView()
        }
    }
}

struct EnvironmentObjectSubView:View {
    @EnvironmentObject var settings:DataModelEnvironmentObject
    @EnvironmentObject var settings1:DataModelEnvironmentObject
    var body: some View {
        Text("SubView")
        Text("count: \(settings.count)")
        Button("点击修改") {
            settings.count+=1
            settings1.count+=1
        }
    }
}

// -------------------------------
// @Environment

// 扩展自定义系统环境变量
extension EnvironmentValues {
    @Entry var myCustomValue: String = "Default value"
}

struct EnvironmentMainView:View {
    @Environment(\.colorScheme) var colorScheme:ColorScheme
    @Environment(\.myCustomValue) var customValue:String
    var body: some View {
        if colorScheme == .light {
            Text("light")
        }
        EnvironmentSubView()
            .environment(\.myCustomValue,"New value")
    }
}

struct EnvironmentSubView:View {
    @Environment(\.colorScheme) var colorScheme:ColorScheme
    @Environment(\.myCustomValue) var customValue:String
    var body: some View {
        Text(customValue)
        if colorScheme == .dark {
            Text("dark")
        }
    }
}

// ---------------------------------------
// @Observable 宏 17.5
// @State
// @Environment


struct Book:Identifiable {
    var id = UUID()
    var name:String
    var content:String
}

@available(iOS 17.0, *)
@Observable
class Library {
    var count = 0
    var books:[Book] = [Book(name: "IOS", content: "讲述IOS"),
                        Book(name: "JAVASCRIPT", content: "讲述JAVASCRIPT")]
}

@available(iOS 17.0, *)
struct ObservableMainView:View {
    @State var library = Library()
    var body: some View {
        ObservableSubView()
            .environment(library)
    }
}

@available(iOS 17.0, *)
struct ObservableSubView:View {
    @Environment(Library.self) private var library
    @State private var titleField:String = ""
    @State private var contentField:String = ""
    var body: some View {
        Text("\(library.count)")
        VStack{
            ForEach(library.books) { book in
                HStack {
                    Text("标题：\(book.name)")
                    Text("内容：\(book.content)")
                }
            }
        }
       
        HStack{
            Button(action:{
                library.count += 1
            }){
                Text("修改值")
                    .frame(maxWidth: .infinity,maxHeight: 45)
                    .background(.red)
                    .cornerRadius(8)
                    .overlay {
                        RoundedRectangle(cornerRadius: 8)
                            .stroke(.blue, lineWidth: 2)
                            .foregroundStyle(.white)
                            .opacity(0.5)
                    }
            }
        }.padding(.init(top: 0, leading: 20, bottom: 0, trailing: 20))
        TextField("标题", text: $titleField)
        TextField("内容", text: $contentField)
        HStack{
            Button(action:{
                guard let title = Optional(titleField),let content = Optional(contentField) else {return}
                library.books.append(Book(name: title, content: content))
            }){
                Text("添加书")
                    .frame(maxWidth: .infinity,maxHeight: 45)
                    .background(.red)
                    .cornerRadius(8)
                    .overlay {
                        RoundedRectangle(cornerRadius: 8)
                            .stroke(.blue, lineWidth: 2)
                            .foregroundStyle(.white)
                            .opacity(0.5)
                    }
            }
        }.padding(.init(top: 0, leading: 20, bottom: 0, trailing: 20))
       
    }
}
